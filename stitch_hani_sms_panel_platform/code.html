Runs within a PostgreSQL serializable transaction isolation bracket to prevent race-condition concurrency overages during bulk invitations. 
import { NextResponse } from 'next/server';
import { getServerSession } from 'next-auth';
import { prisma } from '@/lib/prisma';
import { authOptions } from '@/lib/auth';
import bcrypt from 'bcryptjs';

// STRICT 5-MEMBER CAP ENFORCED BY SERIALIZABLE ISOLATION
const MAX_TEAM_SEATS = 5;

export async function POST(req: Request) {
  const session = await getServerSession(authOptions);

  if (!session?.user?.id) {
    return NextResponse.json({ error: 'UNAUTHENTICATED' }, { status: 401 });
  }

  if (!['ADMIN', 'SUPER_ADMIN'].includes(session.user.role)) {
    return NextResponse.json({ error: 'FORBIDDEN_PRIVILEGE_REQUIRED' }, { status: 403 });
  }

  const { name, email, password } = await req.json();

  if (!email || !password) {
    return NextResponse.json({ error: 'EMAIL_AND_PASSWORD_REQUIRED' }, { status: 400 });
  }

  try {
    const newMember = await prisma.$transaction(async (tx) => {
      // 1. Lock and count existing allocated seats for this parent account
      const allocatedSeats = await tx.user.count({
        where: {
          parentId: session.user.id,
          role: 'TEAM_MEMBER',
        },
      });

      if (allocatedSeats >= MAX_TEAM_SEATS) {
        throw new Error('LIMIT_EXCEEDED_5_SEATS_MAX');
      }

      // 2. Deduplication check
      const exists = await tx.user.findUnique({ where: { email } });
      if (exists) {
        throw new Error('EMAIL_ALREADY_REGISTERED');
      }

      // 3. Secure Hash and Provision
      const passwordHash = await bcrypt.hash(password, 12);

      return await tx.user.create({
        data: {
          name,
          email,
          passwordHash,
          role: 'TEAM_MEMBER',
          status: 'APPROVED',
          parentId: session.user.id,
        },
        select: {
          id: true,
          name: true,
          email: true,
          role: true,
          status: true,
          createdAt: true,
        },
      });
    }, {
      isolationLevel: 'Serializable',
      maxWait: 5000,
      timeout: 10000,
    });

    return NextResponse.json({ data: newMember, success: true }, { status: 201 });
  } catch (err: any) {
    if (err.message === 'LIMIT_EXCEEDED_5_SEATS_MAX') {
      return NextResponse.json({ error: 'Tier Cap: Maximum 5 team members permitted per account' }, { status: 400 });
    }
    return NextResponse.json({ error: err.message || 'INTERNAL_SERVER_ERROR' }, { status: 500 });
  }
}

import NextAuth, { NextAuthOptions } from 'next-auth';
import CredentialsProvider from 'next-auth/providers/credentials';
import { prisma } from '@/lib/prisma';
import bcrypt from 'bcryptjs';

export const authOptions: NextAuthOptions = {
  session: { strategy: 'jwt', maxAge: 30 * 24 * 60 * 60 },
  pages: {
    signIn: '/auth/login',
    error: '/auth/status-pending',
  },
  providers: [
    CredentialsProvider({
      name: 'Credentials',
      credentials: {
        email: { label: 'Email', type: 'email' },
        password: { label: 'Password', type: 'password' },
      },
      async authorize(credentials) {
        if (!credentials?.email || !credentials?.password) return null;

        const user = await prisma.user.findUnique({
          where: { email: credentials.email },
        });

        if (!user || !(await bcrypt.compare(credentials.password, user.passwordHash))) {
          throw new Error('INVALID_CREDENTIALS');
        }

        // STRICT STATUS GATE: Block access if pending admin review
        if (user.status === 'PENDING') {
          throw new Error('ACCOUNT_PENDING_APPROVAL');
        }
        if (user.status === 'REJECTED') {
          throw new Error('ACCOUNT_ACCESS_REVOKED');
        }

        return {
          id: user.id,
          email: user.email,
          name: user.name,
          role: user.role,
          status: user.status,
          apiKey: user.apiKey,
        };
      },
    }),
  ],
  callbacks: {
    async jwt({ token, user }) {
      if (user) {
        token.id = user.id;
        token.role = (user as any).role;
        token.status = (user as any).status;
        token.apiKey = (user as any).apiKey;
      }
      return token;
    },
    async session({ session, token }) {
      if (session.user) {
        (session.user as any).id = token.id;
        (session.user as any).role = token.role;
        (session.user as any).status = token.status;
        (session.user as any).apiKey = token.apiKey;
      }
      return session;
    },
  },
};
